import turtle as t
import time

def quadrat():
    for _ in range(4):
        t.forward(50)
        t.right(90)

def drehen():
    runden =int(input("Wie viele Runden?"))
    for _ in range(runden):
        for _ in range(360):
            t.width(1)
            t.color("black")
            quadrat()
            time.sleep(0.001)
            t.width(10)
            t.color("white")
            quadrat()
            t.right(1)

t.hideturtle()
drehen()

import turtle as t
import time

def quadrat():
    for _ in range(4):
        t.forward(50)
        t.right(90)

def drehen():
    runden =int(input("Wie viele Runden?"))
    for _ in range(runden):
        for _ in range(360):
            # quadrat malen
            t.width(1)
            t.color("black")
            quadrat()
            time.sleep(0.001) # 1 ms warten
            # quadrat auslöschen
            t.width(10)
            t.color("white")
            quadrat()
            # nach rechts drehen
            t.right(1)

t.hideturtle()
drehen()
