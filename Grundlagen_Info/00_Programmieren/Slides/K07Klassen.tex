\begin{frame}[fragile]{Vordefinierte Klassen in Python}
	\begin{itemize}[<+->]

	\end{itemize}

\begin{frame}[fragile]{Klassenattribute vs. Instanzattribute}
	\begin{itemize}[<+->]
		\item \textbf{Instanzattribute} (\lstinline|self.attribut|):
		      Gelten individuell für ein einzelnes Objekt (z.B. \lstinline|self.name|, \lstinline|self.alter|).
		\item \textbf{Klassenattribute}:
		      Werden direkt in der Klasse definiert und sind für \textbf{alle Instanzen gleich}.
	\end{itemize}
	\pause
	\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
class Schueler:
    schule = "Kantonsschule im Lee" # Klassenattribut fuer alle

    def __init__(self, name, klasse):
        self.name = name            # Instanzattribut
        self.klasse = klasse        # Instanzattribut

s1 = Schueler("Lisa", "3a")
s2 = Schueler("Tim", "4b")
print(s1.schule) # Ausgabe: Kantonsschule im Lee

Schueler.schule = "KLW" # Aenderung wirkt auf alle Instanzen!
print(s2.schule) # Ausgabe: KLW
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Instanzzähler \& \lstinline|@classmethod|}
	\begin{itemize}
		\item Klassenattribute eignen sich ideal zum \textbf{Zählen von Instanzen}.
		\item Mit \lstinline|@classmethod| definieren wir Methoden, die sich auf die Klasse (\lstinline|cls|) beziehen.
	\end{itemize}
	\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
class Produkt:
    anzahl_produkte = 0 # Klassenattribut

    def __init__(self, name, preis):
        self.name = name
        self.preis = preis
        Produkt.anzahl_produkte += 1 # Bei jeder neuen Instanz +1

    @classmethod
    def get_anzahl_produkte(cls):
        print("Anzahl Produkte:", cls.anzahl_produkte)

p1 = Produkt("Laptop", 1200)
p2 = Produkt("Maus", 30)
Produkt.get_anzahl_produkte() # Ausgabe: Anzahl Produkte: 2
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Vererbung (Inheritance)}
	\begin{block}{Definition Vererbung}
		Bei der \textbf{Vererbung} erbt eine Kindklasse (Subklasse) Attribute und Methoden von einer Elternklasse (Superklasse). Sie kann eigene Attribute/Methoden hinzufügen oder bestehende Methoden überschreiben.
	\end{block}
	\pause
	\begin{itemize}[<+->]
		\item Syntax für Vererbung: \lstinline|class Kindklasse(Elternklasse):|
		\item Mit \lstinline|super().__init__(...)| wird der Konstruktor der Elternklasse aufgerufen.
		\item Vermeidet Code-Duplizierung und ermöglicht modulare Programmierstrukturen.
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Beispiel: Vererbung}
	\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
class Fahrzeug:
    def __init__(self, marke, modell):
        self.marke = marke
        self.modell = modell
    def info(self):
        return f"{self.marke} {self.modell}"

class ElektroAuto(Fahrzeug): # ElektroAuto erbt von Fahrzeug
    def __init__(self, marke, modell, batterie_kapazitaet):
        super().__init__(marke, modell) # Eltern-Konstruktor
        self.batterie_kapazitaet = batterie_kapazitaet

    def info(self): # Methode der Elternklasse ueberschreiben
        basis = super().info()
        return f"{basis}, Kapazitaet: {self.batterie_kapazitaet} kWh"

e_car = ElektroAuto("Tesla", "Model 3", 75)
print(e_car.info()) # Tesla Model 3, Kapazitaet: 75 kWh
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Polymorphismus (\enquote{Viele Formen})}
	\begin{block}{Definition Polymorphismus}
		Verschiedene Klassen können Methoden mit \textbf{demselben Namen} besitzen, aber jeweils eine eigene, spezifische Implementierung haben (\textit{poly} = viel, \textit{morphos} = Form).
	\end{block}
	\pause
	\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
class Tier:
    def geraeusch(self):
        return "Ein Tier macht ein Geraeusch"

class Hund(Tier):
    def geraeusch(self):
        return "Wuff!"

class Katze(Tier):
    def geraeusch(self):
        return "Miau!"

tiere = [Tier(), Hund(), Katze()]
for t in tiere:
    print(t.geraeusch()) # Jedes Objekt ruft seine eigene Methode auf!
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Praktisches Beispiel: Bibliothekssystem}

		\item In komplexen Anwendungen arbeiten \textbf{mehrere Klassen zusammen}:

class Bibliothek:
    def __init__(self, name):
        self.name = name
        self.buecher = []
        self.mitglieder = []

    def buch_hinzufuegen(self, buch):
        self.buecher.append(buch)
        print(f"Buch '{buch.titel}' zur Bibliothek hinzugefuegt.")
\end{lstlisting}

\begin{frame}[fragile]{Zusammenfassung: Klassen \& OOP}
	\begin{itemize}[<+->]
		\item \textbf{Klasse}: Bauplan für eigene Datentypen mit Attributen und Methoden.
		\item \textbf{Instanz / Objekt}: Konkretes Exemplar (\lstinline|self|).
		\item \textbf{Klassenattribute \& \lstinline|@classmethod|}: Für alle Instanzen einer Klasse gemeinsam.
		\item \textbf{Vererbung}: Kindklassen erben Eigenschaften von Elternklassen (\lstinline|super()|).
		\item \textbf{Polymorphismus}: Gleicher Methodenname, unterschiedliches Verhalten.
	\end{itemize}
\end{frame}
