\begin{frame}{Huffman-Kodierung}
\begin{itemize}[<+->]
\item Optimale Kodierung von Zeichen.
\item Wenn die Häufigkeiten der Zeichen bei 1/2, 1/4, 1/8, etc. liegen, gibt es keine bessere Kodierung
\item Jedes Zeichen erhält eine ganze Zahl von Bits: 1 Bit passt zu 50\%, 2 Bits zu 25\%
\item Bei 30\% wäre eine Länge zwischen 1 und 2 Bits passend, Huffman verschenkt Platz
\end{itemize}
\end{frame}

\begin{frame}{Jorma Rissanen}
\begin{itemize}[<+->]
\item Ganze Texte anstatt einzelne Buchstaben kodieren!
\item Idee: Jedem Text wird eine Zahl zwischen 0 und 1 zugeordnet
\item Je länger der Text, desto mehr Nachkommastellen
\item Ist unter gewissen Umständen kürzer als Huffman
\end{itemize}
\end{frame}

\begin{frame}{Vom Intervall zur Bitfolge}
\texttt{ADBA} $\rightarrow$ Intervall $[0.475, 0.48)$
\begin{itemize}[<+->]
\item Gespeichert wird eine Zahl im Intervall mit möglichst wenigen Nachkommastellen im Zweiersystem
\item 6 Stellen (Vielfache von $\frac{1}{64}$): $\frac{30}{64}=0.46875$ zu klein, $\frac{31}{64}=0.484375$ zu gross
\item 7 Stellen: $\frac{61}{128}=0.4765625=0.0111101_2$ $\rightarrow$ 7 Bits \texttt{0111101} und Textlänge 4
\item Huffman (A $=$ \texttt{0}, C $=$ \texttt{10}, B $=$ \texttt{110}, D $=$ \texttt{111}): 8 Bits
\end{itemize}
\end{frame}
\begin{frame}{Dekodieren}
Zahl $0.4765625$, Textlänge 4: Zeile für Zeile das Teilintervall suchen
\begin{itemize}[<+->]
\item $[0, 0.5)$ $\rightarrow$ \texttt{A}
\item $[0.45, 0.5)$ $\rightarrow$ \texttt{AD}
\item $[0.475, 0.485)$ $\rightarrow$ \texttt{ADB}
\item $[0.475, 0.48)$ $\rightarrow$ \texttt{ADBA}, nach 4 Buchstaben aufhören
\item Ohne Textlänge unklar: Die Zahl liegt auch im Intervall von \texttt{ADBAA}
\item Ist ein Intervall gegeben: dekodieren, bis das Intervall übereinstimmt
\end{itemize}
\end{frame}
\begin{frame}{Nachteil von arithmetischer Kodierung}
Komprimieren und Dekomprimieren ist mit Huffman effizienter $\rightarrow$ Abwägen (\textit{trade-off}) zwischen Speicherplatz und Rechenaufwand
\end{frame}
