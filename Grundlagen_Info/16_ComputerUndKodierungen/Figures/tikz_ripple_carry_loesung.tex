    \node[fa] (fa0) at (7.2, 0) {\textbf{FA 0}\\[-2pt]{\scriptsize (Bit 0)}};

    % Inputs from top (a_i in NavyBlue, b_i in BrickRed)
    \draw[thick, ->, NavyBlue] (-0.5, 2.1) node[above, font=\footnotesize, NavyBlue] {$a_2 = 0$} -- (-0.5, 0.8);
