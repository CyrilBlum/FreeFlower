\begin{frame}\FDUtitle{Der Arbeitsspeicher (RAM)}
\begin{center}
Stellen Sie sich RAM als eine lange Tabelle vor: jede Zeile ist eine Speicherzelle mit eindeutiger Adresse.
\end{center}
\begin{table}[H]
    \centering
    \begin{tabular}{cc}
        \textbf{Adresse} & \textbf{Inhalt (1 Byte)} \\\midrule
        \texttt{0x00} & \texttt{10110100} \\
        \texttt{0x01} & \texttt{00001111} \\
        \texttt{0x02} & \texttt{11001010} \\
        $\vdots$ & $\vdots$ \\
    \end{tabular}
\end{table}
\begin{center}
Die Adressen sind hexadezimal (Lektion 3) -- daher das Präfix \texttt{0x}.
\end{center}
\end{frame}

\begin{frame}\FDUtitle{RAM vs.\ Cache}
\begin{itemize}[<+->]\setlength\itemsep{7pt}
  \item RAM ist \tib{flüchtig}: Inhalt geht beim Ausschalten verloren
  \item Typische Grösse: 8--64 GB
  \item Der \tib{Cache}: noch schneller, aber viel kleiner -- speichert häufig genutzte Daten zwischen
\end{itemize}
\end{frame}

\begin{frame}\FDUtitle{Busschnittstellen: \gls{PCIe} \& Lanes}
\begin{itemize}\setlength\itemsep{4pt}
  \item \textbf{\gls{PCIe}} (\emph{Peripheral Component Interconnect Express}): modularer Hochgeschwindigkeits-Standard auf dem Mainboard.
  \item Parallele Datenbahnen, sogenannte \textbf{Lanes} ($\times 1, \times 4, \times 8, \times 16$).
  \item Die Bandbreite verdoppelt sich mit jeder Generation:
\end{itemize}
\begin{center}\small
\begin{tabular}{lrrr}
    \textbf{Generation} & \textbf{Lane ($\times 1$)} & \textbf{$\times 4$ (z.\,B.\ M.2-SSD)} & \textbf{$\times 16$ (z.\,B.\ GPU)} \\\midrule
    PCIe 3.0 & ca.\ $1\,\mathrm{GB/s}$ & ca.\ $4\,\mathrm{GB/s}$ & ca.\ $16\,\mathrm{GB/s}$ \\
    PCIe 4.0 & ca.\ $2\,\mathrm{GB/s}$ & ca.\ $8\,\mathrm{GB/s}$ & ca.\ $32\,\mathrm{GB/s}$ \\
    PCIe 5.0 & ca.\ $4\,\mathrm{GB/s}$ & ca.\ $16\,\mathrm{GB/s}$ & ca.\ $64\,\mathrm{GB/s}$ \\
\end{tabular}
\end{center}
\vspace{1mm}
\footnotesize Zum Vergleich: Ein klassisches \textbf{SATA}-Kabel schafft maximal ca.\ $0{,}55\,\mathrm{GB/s}$ ($550\,\mathrm{MB/s}$).
\end{frame}
