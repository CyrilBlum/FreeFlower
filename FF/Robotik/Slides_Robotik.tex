\begin{frame}{Der Analog-Digital-Wandler (ADC)}
	\begin{itemize}[<+->]
		\item Ein Computer kennt nur Bits ($0$ und $1$). 
		\item Ein ADC (Analog-to-Digital Converter) misst eine Spannung und wandelt sie in eine Zahl um.
		\item \textbf{Beispiel 10-Bit Auflösung:}
		\begin{itemize}
			\item $0$ Volt $\rightarrow$ Zahl $0$
			\item $3.3$ Volt $\rightarrow$ Zahl $1023$ ($2^{10} - 1$)
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Eingabe messen}
	\begin{myexercise}
    Stellen Sie sich vor, wir lesen ein Potentiometer mit einem 10-Bit ADC aus. Der gelesene Wert ist $512$.
    
    Welche Spannung liegt (bei einer Referenz von $3.3$V) ungefähr am Pin an?
	\end{myexercise}
\end{frame}

\begin{frame}
\begin{myanswer}
    Da $512$ genau die Hälfte von $1024$ ist, liegt die Spannung bei der Hälfte der Referenzspannung:
    \[ \frac{512}{1024} \cdot 3.3V = 1.65V \]
\end{myanswer}
\end{frame}

\begin{frame}{Ausgabe: Wie dimmen wir eine LED?}
	\begin{itemize}[<+->]
		\item Ein digitaler Pin kann nur \textbf{AN} ($3.3$V) oder \textbf{AUS} ($0$V).
		\item Wir können die Spannung am Pin meist \textit{nicht} direkt senken.
		\item \textbf{Trick:} Pulsweitenmodulation (PWM).
		\item Wir schalten die LED so schnell an und aus, dass das menschliche Auge nur die \enquote{mittlere} Helligkeit wahrnimmt.
	\end{itemize}
\end{frame}

\begin{frame}{PWM: Duty Cycle (Tastgrad)}
	\begin{itemize}[<+->]
		\item \textbf{10\% Duty Cycle:} Kurz AN, lange AUS $\rightarrow$ Dunkel.
		\item \textbf{50\% Duty Cycle:} Gleich lang AN wie AUS $\rightarrow$ Mittlere Helligkeit.
		\item \textbf{90\% Duty Cycle:} Fast immer AN $\rightarrow$ Hell.
	\end{itemize}
	\vspace{0.5cm}
	\begin{center}
		\begin{tikzpicture}[scale=0.7]
			% 25% PWM
			\draw[thick] (0,1) -- (0.5,1) -- (0.5,0) -- (2,0) -- (2,1) -- (2.5,1) -- (2.5,0) -- (4,0);
			\node at (-1,0.5) {25\%};
			% 75% PWM
			\draw[thick] (6,1) -- (7.5,1) -- (7.5,0) -- (8,0) -- (8,1) -- (9.5,1) -- (9.5,0) -- (10,0);
			\node at (5,0.5) {75\%};
		\end{tikzpicture}
	\end{center}
\end{frame}

\begin{frame}{Der Algorithmus: Was macht eine Kerze aus?}
	\begin{itemize}[<+->]
		\item \textbf{Basis-Helligkeit:} Ein konstanter Lichtstrom (eingestellt via Poti).
		\item \textbf{Rauschen:} Kleine, schnelle Änderungen der Helligkeit.
		\item \textbf{Flackern:} Gelegentliche, grössere \enquote{Sprünge} nach unten (Windstoss).
	\end{itemize}
\end{frame}

\begin{frame}[fragile]
\begin{myanswer}
    Mögliche Umsetzung in MicroPython:
\begin{lstlisting}[language=Python]
import machine, time, random

led = machine.PWM(machine.Pin(15))
led.freq(1000)

while True:
    # Erzeuge zufaellige Helligkeit (flackern)
    brightness = random.randint(20, 150) 
    led.duty_u16(brightness * 256) # auf 16-bit skalieren
    
    # Zufallige kurze Pause
    time.sleep_ms(random.randint(10, 100))
\end{lstlisting}
\end{myanswer}
\end{frame}
